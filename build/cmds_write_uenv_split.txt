echo ===WRITE_UENV=== >> /dev/ttyPS0
cd /run/media/mmcblk0p1
printf '%s\n' 'bootargs=console=ttyPS0,115200 earlyprintk mem=256M clk_ignore_unused cpuidle.off=1 idle=poll maxcpus=1 usbcore.autosuspend=-1' > uEnv.txt
printf '%s\n' 'bootfx=mw.l 0xE000A20C 0x4000; mw.l 0xE000A210 0x4000; mw.l 0xE000A044 0x4000; mw.l 0xE0002140 0x2' >> uEnv.txt
printf '%s\n' 'splash=if fatload mmc 0:1 0x18000000 hub75_boot.frm; then mw.l 0x41200000 0x18000000; mw.l 0x41200008 0x2; sleep 1; mw.l 0x41200008 0x3; fi' >> uEnv.txt
printf '%s\n' 'uenvcmd=run bootfx; run splash' >> uEnv.txt
sync
echo ===VERIFY=== >> /dev/ttyPS0
ls -l uEnv.txt >> /dev/ttyPS0 2>&1
cat uEnv.txt >> /dev/ttyPS0 2>&1
md5sum uEnv.txt >> /dev/ttyPS0 2>&1
echo ===DONE=== >> /dev/ttyPS0
