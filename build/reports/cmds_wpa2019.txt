dd if=/mnt/boot/uramdisk.image.gz of=/tmp/ramdisk.gz bs=64 skip=1
gunzip -c /tmp/ramdisk.gz > /tmp/demo.ext2
mkdir -p /tmp/demo
mount -o loop -t ext2 /tmp/demo.ext2 /tmp/demo
ls -l /tmp/demo/sbin/wpa_supplicant
printf '#!/bin/sh
exec /tmp/demo/lib/ld-linux.so.3 --library-path /tmp/demo/lib:/tmp/demo/usr/lib /tmp/demo/sbin/wpa_supplicant "$@"
' > /tmp/wp.sh
chmod +x /tmp/wp.sh
mkdir -p /var/run/wpa_supplicant
printf 'ctrl_interface=/var/run/wpa_supplicant
ap_scan=1
network={
ssid="AllianceTeam2.4G"
psk="12345678"
key_mgmt=WPA-PSK
}
' > /tmp/wpa.conf
cat /tmp/wpa.conf
echo MARK_WPA2019
/tmp/wp.sh -Dwext -iwlan0 -c/tmp/wpa.conf > /tmp/wpa.log 2>&1 &
sleep 7
cat /tmp/wpa.log
ps | grep wpa
echo MARK_WPA2019_DONE
