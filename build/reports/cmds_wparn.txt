md5sum /tmp/wpa_supplicant /tmp/wpa_cli
chmod +x /tmp/wpa_supplicant /tmp/wpa_cli
mkdir -p /var/run/wpa_supplicant
cat > /tmp/wpa.conf << 'EOF'
ctrl_interface=/var/run/wpa_supplicant
ap_scan=1
network={
ssid="AllianceTeam2.4G"
psk="12345678"
}
EOF
cat /tmp/wpa.conf
echo MARK_WPARN
/tmp/wpa_supplicant -Dnl80211 -iwlan0 -c/tmp/wpa.conf > /tmp/wpa.log 2>&1 &
sleep 8
cat /tmp/wpa.log
ps | grep wpa
echo MARK_WPARN_DONE
